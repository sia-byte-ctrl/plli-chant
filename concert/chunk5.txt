function renderOverview(){
  const root=document.getElementById("overview");
  root.innerHTML=days.map(([date,label,desc,cls])=>{

    return `<section class="day">
      <div class="dayhead ${cls}">
        <div class="date">${label}</div>
        <div class="desc">${desc}</div>
      </div>
      <div class="list">

          const names=(it.people||[]).map(p=>peopleName[p]).join(" · ");

            title="눌러서 수정"
          >
            <div class="time">${esc(it.time)}</div>
            <div class="content">
              <div class="title">${esc(it.title)}</div>
              <div class="chips">
                ${names ? `<span class="chip ${isVip(it)?"vip":""}">${esc(names)}</span>` : ""}

      </div>
    </section>`
  }).join("");
}

function copyAddress(text){
  if(navigator.clipboard){
    navigator.clipboard.writeText(text).then(()=>alert("주소를 복사했어."));
  }else{
    prompt("주소를 복사해줘.",text);
  }
}

function renderParking(){
  if(!data.parking) data.parking={name:"",address:"",fee:"",status:"미정",note:""};
  const p=data.parking;
  const set=(id,value,fallback="미정")=>{
    const el=document.getElementById(id);
    if(!el)return;
    el.textContent=value||fallback;
    el.classList.toggle("parking-empty",!value);
  };
  set("parkingName",p.name);
  set("parkingAddress",p.address);
  set("parkingFee",p.fee);
  const note=document.getElementById("parkingNote");
  if(note){
    const status=p.status||"미정";
    note.textContent=p.note ? `${status} · ${p.note}` : `${status} · 나중에 정하면 여기서 바로 입력하면 돼.`;
  }
}
function openParkingEditor(){
  if(!data.parking) data.parking={name:"",address:"",fee:"",status:"미정",note:""};
  pName.value=data.parking.name||"";
  pAddress.value=data.parking.address||"";
  pFee.value=data.parking.fee||"";
  pStatus.value=data.parking.status||"미정";
  pNote.value=data.parking.note||"";
  parkingDialog.showModal();
}
function saveParking(){
  data.parking={
    name:pName.value.trim(),
    address:pAddress.value.trim(),
    fee:pFee.value.trim(),
    status:pStatus.value,
    note:pNote.value.trim()
  };
  save();
  renderParking();
  parkingDialog.close();
}

function ensureChecklistData(){
  if(!data.checklist || typeof data.checklist!=="object") data.checklist=clone(defaultData.checklist);
  if(!data.checked || typeof data.checked!=="object") data.checked={};
  if(!Array.isArray(data.customChecklist)) data.customChecklist=[];
}
function isCustomSupply(cat,item){
  ensureChecklistData();
  return data.customChecklist.some(x=>x.cat===cat && x.item===item);
}
function renderChecklist(){

  const root=document.getElementById("checkgrid");

      ${items.map(item=>{
