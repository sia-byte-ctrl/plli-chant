function moveItem(id,dir){
  const i=data.schedules.findIndex(x=>x.id===id);if(i<0)return;const j=i+dir;

  [data.schedules[i],data.schedules[j]]=[data.schedules[j],data.schedules[i]];save();render();
}

function closeDialogOnBackdrop(dialogEl){
  dialogEl.addEventListener("click",(e)=>{
    const rect=dialogEl.getBoundingClientRect();
    const outside=e.clientX<rect.left||e.clientX>rect.right||e.clientY<rect.top||e.clientY>rect.bottom;
    if(outside) dialogEl.close();
  });
}
