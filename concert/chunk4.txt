async function loadCloud(){
  try{
    const url=CLOUD_ENDPOINT;
    const res=await fetch(url,{method:"GET",cache:"no-store"});
    if(!res.ok) throw new Error("cloud read failed");
    const payload=await res.json();
    if(payload && payload.data && Array.isArray(payload.data.schedules) && payload.data.schedules.length){
      data=payload.data;
      localStorage.setItem(KEY,JSON.stringify(data));
      render();
    }else{
      await saveCloudNow();
    }
    cloudReady=true;
    document.getElementById("saved").textContent="클라우드 저장됨 ✓";
  }catch(e){
    cloudReady=false;
    document.getElementById("saved").textContent="오프라인 저장 중";
  }
}
function save(){
  document.getElementById("saved").textContent="저장 중…";
  localStorage.setItem(KEY,JSON.stringify(data));
  clearTimeout(cloudSaveTimer);
  cloudSaveTimer=setTimeout(saveCloudNow,350);
}
async function saveCloudNow(){
  try{
    const url=CLOUD_ENDPOINT;
    const res=await fetch(url,{
      method:"POST",
      headers:{"Content-Type":"application/json"},
      body:JSON.stringify({data})
    });
    if(!res.ok) throw new Error("cloud write failed");
    cloudReady=true;
    document.getElementById("saved").textContent="클라우드 저장됨 ✓";
  }catch(e){
    cloudReady=false;
    document.getElementById("saved").textContent="오프라인 저장 중";
  }
}
