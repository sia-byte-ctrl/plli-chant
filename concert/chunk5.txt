function copyAddress(text){
  if(navigator.clipboard){
    navigator.clipboard.writeText(text).then(()=>alert("주소를 복사했어."));
  }else{
    prompt("주소를 복사해줘.",text);
  }
}

function renderParking(){
  if(!data.parking) data.parking={name:"",address:"",fee:"",status:"미정",note:""};
  const p=data.parking;
  const set=(id,value,fallback="미정")=>{
    const el=document.getElementById(id);
    if(!el)return;
    el.textContent=value||fallback;
    el.classList.toggle("parking-empty",!value);
  };
  set("parkingName",p.name);
  set("parkingAddress",p.address);
  set("parkingFee",p.fee);
  const note=document.getElementById("parkingNote");
  if(note){
    const status=p.status||"미정";
    note.textContent=p.note ? `${status} · ${p.note}` : `${status} · 나중에 정하면 여기서 바로 입력하면 돼.`;
  }
}
